import argparse
import csv
import json
from pathlib import Path

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", type=Path, required=True)
    parser.add_argument("--output", type=Path)
    parser.add_argument("--preflight", action="store_true")
    args = parser.parse_args()
    config = json.loads(args.config.read_text(encoding="utf-8"))
    capital = float(config["initial_capital"])
    if capital <= 0:
        raise ValueError("initial_capital must be positive")
    if args.preflight:
        print(json.dumps({"schema_version":"quant-studio.custom-preflight/v1", "software_preflight":"pass", "rows":3, "read_only":True, "investable":False}))
        return
    if args.output is None:
        raise ValueError("--output required")
    args.output.mkdir(parents=True, exist_ok=False)
    with (args.output / "nav.csv").open("w", newline="", encoding="utf-8") as stream:
        writer = csv.writer(stream)
        writer.writerow(["date","nav","initial_nav"])
        for date, value in [("2025-01-02",capital),("2025-01-03",capital*1.01),("2025-01-06",capital*0.995)]:
            writer.writerow([date,value,capital])
    (args.output / "report.html").write_text("<h1>Custom module example</h1><p>Synthetic data. Replace with your research implementation.</p>", encoding="utf-8")

if __name__ == "__main__":
    main()
